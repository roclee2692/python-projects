#include <stdio.h>

int pre[1010], post[1010];
int n, cnt = 0;

void getPost(int root, int end, int mirror, int *ok)
{
    if(root > end || *ok == 0) return;
    if(root == end)
    {
        post[cnt++] = pre[root];
        return;
    }

    int i, j;
    int r = pre[root];

    if(!mirror)   /* 普通BST */
    {
        i = root + 1;
        while(i <= end && pre[i] < r) i++;
        j = i;
        while(j <= end)
        {
            if(pre[j] < r)
            {
                *ok = 0;
                return;
            }
            j++;
        }
        getPost(root + 1, i - 1, mirror, ok);
        getPost(i, end, mirror, ok);
        post[cnt++] = r;
    }
    else          /* 镜像BST */
    {
        i = root + 1;
        while(i <= end && pre[i] >= r) i++;
        j = i;
        while(j <= end)
        {
            if(pre[j] >= r)
            {
                *ok = 0;
                return;
            }
            j++;
        }
        getPost(root + 1, i - 1, mirror, ok);
        getPost(i, end, mirror, ok);
        post[cnt++] = r;
    }
}

int main()
{
    int i;
    scanf("%d", &n);
    for(i = 0; i < n; i++)
        scanf("%d", &pre[i]);

    int ok = 1;
    cnt = 0;
    getPost(0, n - 1, 0, &ok);

    if(!ok)
    {
        ok = 1;
        cnt = 0;
        getPost(0, n - 1, 1, &ok);
    }

    if(!ok)
    {
        printf("NO");
    }
    else
    {
        printf("YES\n");
        for(i = 0; i < cnt; i++)
        {
            if(i) printf(" ");
            printf("%d", post[i]);
        }
    }

    return 0;
}
